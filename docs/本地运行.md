## 一次性装配步骤（已在沙箱执行）

```bash
# 1. 数据库
mysql -e "CREATE DATABASE nft DEFAULT CHARSET utf8mb4 COLLATE utf8mb4_unicode_ci;"
mysql -e "CREATE USER 'nft'@'localhost' IDENTIFIED BY '你的密码'; GRANT ALL ON nft.* TO 'nft'@'localhost';"

# 2. 后端依赖
composer install          # 官方 packagist 源（阿里云镜像缺 topthink/framework 8.1.4，勿用）

# 3. 前端依赖
cd web && npm install     # 建议配 npmmirror

# 4. 数据库配置（.env，PREFIX 留空）
# [DATABASE] HOSTNAME=127.0.0.1 DATABASE=nft USERNAME=nft PASSWORD=... PREFIX= CHARSET=utf8mb4

# 5. 建表：先框架系统表（迁移），再业务表
php think migrate:run
mysql nft < database/nft.sql

# 6. 管理员密码（种子默认密码未知，重置一次）
#    临时脚本：new \app\admin\model\Admin())->resetPassword(1, '123456');

# 7. 前端构建 + 发布
cd web && npm run build
cp -r web/dist/index.html web/dist/assets public/
echo 'install-end' > public/install.lock

# 8. 启动
php think run --host 0.0.0.0 --port 8787
# 访问 http://127.0.0.1:8787/admin  （admin / 123456）
```
